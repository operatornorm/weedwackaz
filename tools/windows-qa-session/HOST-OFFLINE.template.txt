& {
    $ErrorActionPreference = 'Stop'
    Set-StrictMode -Version 2.0
    if ([Security.Principal.WindowsIdentity]::GetCurrent().Name -match '\\WDAGUtilityAccount$') { throw 'Run this block on your normal Windows desktop, OUTSIDE Sandbox.' }
    $mgsvExpectedRun = '__EXPECTED_RUN__'
    $mgsvExpectedSession = '__EXPECTED_SESSION__'
    $mgsvPreloginTag = '__EXPECTED_PRELOGIN_TAG__'
    $mgsvOfflineTag = '__EXPECTED_OFFLINE_SHA256__'
    if ($mgsvExpectedRun -notmatch '^[0-9a-f]{32}$' -or $mgsvExpectedSession -notmatch '^[0-9a-f]{32}$' -or $mgsvOfflineTag -notmatch '^[0-9a-f]{64}$') { throw 'This template must be filled by the package builder first.' }
    $mgsvRoot = Join-Path ([Environment]::GetFolderPath('UserProfile')) ('Downloads\MGSV-QA-' + $mgsvExpectedRun)
    $mgsvInput = Join-Path $mgsvRoot 'Input'
    $mgsvOutput = Join-Path $mgsvRoot 'Output'
    foreach ($mgsvPath in @($mgsvRoot, $mgsvInput, $mgsvOutput)) {
        $mgsvItem = Get-Item -LiteralPath $mgsvPath -Force
        if (-not $mgsvItem.PSIsContainer -or ($mgsvItem.Attributes -band [IO.FileAttributes]::ReparsePoint)) { throw 'Unexpected QA directory or redirection.' }
    }
    $mgsvManifestPath = Join-Path $mgsvInput 'run.json'
    if ((Get-Item -LiteralPath $mgsvManifestPath).Length -gt 2048) { throw 'Oversized run manifest.' }
    $mgsvManifest = Get-Content -LiteralPath $mgsvManifestPath -Raw | ConvertFrom-Json
    if ($mgsvManifest.schema -ne 1 -or $mgsvManifest.run -ne $mgsvExpectedRun -or $mgsvManifest.prelogin_sha256 -ne $mgsvPreloginTag) { throw 'Wrong QA workspace.' }
    $mgsvGatePath = Join-Path $mgsvOutput 'prelogin.gate'
    if ((Get-Item -LiteralPath $mgsvGatePath).Length -gt 512) { throw 'Oversized prelogin report.' }
    $mgsvExpectedGate = "WWQA1`n$mgsvExpectedRun`n$mgsvExpectedSession`nprelogin`n$mgsvPreloginTag`n1,1,1,1`n"
    if ([IO.File]::ReadAllText($mgsvGatePath) -cne $mgsvExpectedGate) { throw 'Prelogin report does not match the accepted guest session.' }
    $mgsvRequestPath = Join-Path $mgsvInput 'offline-request.json'
    if ((Test-Path -LiteralPath $mgsvRequestPath) -or (Test-Path -LiteralPath (Join-Path $mgsvOutput 'offline.gate'))) { throw 'An offline request/report already exists. Do not overwrite or reuse it.' }
    $mgsvOnlineProfile = Join-Path $mgsvRoot 'MGSV-Test.wsb'
    $mgsvRetiredProfile = Join-Path $mgsvRoot 'MGSV-Test.online-retired'
    if ((Test-Path -LiteralPath $mgsvOnlineProfile) -and (Test-Path -LiteralPath $mgsvRetiredProfile)) { throw 'Both online and retired profiles exist; stop here.' }
    if (-not (Test-Path -LiteralPath $mgsvOnlineProfile) -and -not (Test-Path -LiteralPath $mgsvRetiredProfile)) { throw 'The expected Sandbox profile is missing.' }
    Add-Type -AssemblyName System.Windows.Forms
    $mgsvDialog = [Windows.Forms.OpenFileDialog]::new()
    $mgsvDialog.Title = 'Select GUEST-OFFLINE.txt from the newly extracted offline package'
    $mgsvDialog.Filter = 'Guest offline instructions (GUEST-OFFLINE.txt)|GUEST-OFFLINE.txt'
    $mgsvDialog.InitialDirectory = Join-Path ([Environment]::GetFolderPath('UserProfile')) 'Downloads'
    try {
        if ($mgsvDialog.ShowDialog() -ne [Windows.Forms.DialogResult]::OK) { throw 'Cancelled; nothing changed.' }
        $mgsvSource = $mgsvDialog.FileName
    } finally { $mgsvDialog.Dispose() }
    if ((Get-FileHash -LiteralPath $mgsvSource -Algorithm SHA256).Hash.ToLowerInvariant() -ne $mgsvOfflineTag) { throw 'GUEST-OFFLINE.txt does not match the reviewed package.' }
    $mgsvDestination = Join-Path $mgsvInput 'GUEST-OFFLINE.txt'
    if (Test-Path -LiteralPath $mgsvDestination) {
        if ((Get-FileHash -LiteralPath $mgsvDestination -Algorithm SHA256).Hash.ToLowerInvariant() -ne $mgsvOfflineTag) { throw 'Different guest instructions are already staged.' }
    } else {
        $mgsvRead = [IO.File]::OpenRead($mgsvSource)
        try {
            $mgsvWrite = [IO.File]::Open($mgsvDestination, [IO.FileMode]::CreateNew, [IO.FileAccess]::Write, [IO.FileShare]::None)
            try { $mgsvRead.CopyTo($mgsvWrite); $mgsvWrite.Flush() } finally { $mgsvWrite.Dispose() }
        } finally { $mgsvRead.Dispose() }
    }
    if ((Get-FileHash -LiteralPath $mgsvDestination -Algorithm SHA256).Hash.ToLowerInvariant() -ne $mgsvOfflineTag) { throw 'Staged instruction hash mismatch.' }
    # Rename the launcher only; this does not close or change the running guest.
    # Do not restore/reopen this online profile after any game staging.
    if (Test-Path -LiteralPath $mgsvOnlineProfile) { [IO.File]::Move($mgsvOnlineProfile, $mgsvRetiredProfile) }
    $mgsvRequest = [ordered]@{
        schema = 1
        stage = 'offline-v1'
        run = $mgsvExpectedRun
        session = $mgsvExpectedSession
        challenge = [Guid]::NewGuid().ToString('N')
        source_tag = $mgsvOfflineTag
    }
    $mgsvBytes = [Text.Encoding]::UTF8.GetBytes(($mgsvRequest | ConvertTo-Json))
    $mgsvFile = [IO.File]::Open($mgsvRequestPath, [IO.FileMode]::CreateNew, [IO.FileAccess]::Write, [IO.FileShare]::None)
    try { $mgsvFile.Write($mgsvBytes, 0, $mgsvBytes.Length); $mgsvFile.Flush() } finally { $mgsvFile.Dispose() }
    Write-Output ('Prepared offline instructions in: ' + $mgsvInput)
    Write-Output 'Keep the existing Sandbox OPEN. The online launcher is retired. No host networking was changed.'
    Write-Output 'Next run the GUEST-OFFLINE.txt block INSIDE the same guest, after Steam Go Offline and Steam Exit.'
}

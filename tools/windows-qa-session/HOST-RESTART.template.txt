    function Assert-MgsvPlain($mgsvPath) {
        $mgsvItem = Get-Item -LiteralPath $mgsvPath -Force
        if ($mgsvItem.Attributes -band [IO.FileAttributes]::ReparsePoint) { throw 'Unexpected storage redirection.' }
        return $mgsvItem
    }
    function Convert-MgsvJson($mgsvRaw) {
        # Protocol metadata is deliberately unescaped ASCII. Reject duplicate
        # object keys (including case variants), escapes and unparsed tokens.
        if ($mgsvRaw.Length -gt 16384 -or $mgsvRaw -cmatch '[^\x09\x0a\x0d\x20-\x7e]' -or $mgsvRaw.Contains('\')) { throw 'Invalid protocol text.' }
        $mgsvTokens = [regex]::Matches($mgsvRaw, '"[^"\\\x00-\x1f]*"|true|false|null|-?(?:0|[1-9][0-9]*)(?:\.[0-9]+)?(?:[eE][+-]?[0-9]+)?|[{}\[\]:,]')
        $mgsvStack = [Collections.Generic.Stack[object]]::new()
        $mgsvPosition = 0
        for ($mgsvI = 0; $mgsvI -lt $mgsvTokens.Count; $mgsvI++) {
            $mgsvToken = $mgsvTokens[$mgsvI]
            if ($mgsvRaw.Substring($mgsvPosition, $mgsvToken.Index - $mgsvPosition).Trim().Length -ne 0) { throw 'Unparsed protocol token.' }
            $mgsvPosition = $mgsvToken.Index + $mgsvToken.Length
            $mgsvValue = $mgsvToken.Value
            if ($mgsvValue -eq '{') { $mgsvStack.Push(@{kind='object'; seen=@{}}) }
            elseif ($mgsvValue -eq '[') { $mgsvStack.Push(@{kind='array'}) }
            elseif ($mgsvValue -eq '}' -or $mgsvValue -eq ']') {
                if ($mgsvStack.Count -eq 0) { throw 'Unbalanced JSON.' }
                $mgsvTop = $mgsvStack.Pop()
                if (($mgsvValue -eq '}' -and $mgsvTop.kind -ne 'object') -or ($mgsvValue -eq ']' -and $mgsvTop.kind -ne 'array')) { throw 'Wrong JSON container.' }
            } elseif ($mgsvValue.StartsWith('"') -and $mgsvI + 1 -lt $mgsvTokens.Count -and $mgsvTokens[$mgsvI + 1].Value -eq ':') {
                if ($mgsvStack.Count -eq 0 -or $mgsvStack.Peek().kind -ne 'object') { throw 'Invalid object key.' }
                $mgsvKey = $mgsvValue.Substring(1, $mgsvValue.Length - 2)
                $mgsvKeys = $mgsvStack.Peek().seen
                if ($mgsvKeys.ContainsKey($mgsvKey)) { throw 'Duplicate JSON key.' }
                $mgsvKeys[$mgsvKey] = $true
            }
        }
        if ($mgsvStack.Count -ne 0 -or $mgsvRaw.Substring($mgsvPosition).Trim().Length -ne 0) { throw 'Incomplete protocol text.' }
        return (ConvertFrom-Json -InputObject $mgsvRaw)
    }
    function Read-MgsvJson($mgsvPath, $mgsvMaximum) {
        $mgsvItem = Assert-MgsvPlain $mgsvPath
        if ($mgsvItem.PSIsContainer -or $mgsvItem.Length -le 0 -or $mgsvItem.Length -gt $mgsvMaximum) { throw 'Invalid metadata file size.' }
        return (Convert-MgsvJson ([IO.File]::ReadAllText($mgsvPath)))
    }
    function Assert-MgsvKeys($mgsvObject, [string[]]$mgsvExpected) {
        $mgsvKeys = @($mgsvObject.PSObject.Properties | ForEach-Object { $_.Name })
        if ($mgsvKeys.Count -ne $mgsvExpected.Count) { throw 'Wrong metadata property count.' }
        foreach ($mgsvKey in $mgsvKeys) { if ($mgsvKey -cnotin $mgsvExpected) { throw 'Unknown metadata property.' } }
    }
    function Test-MgsvInteger($mgsvValue) { return ($mgsvValue -is [int] -or $mgsvValue -is [long]) }
    function Write-MgsvJson($mgsvPath, $mgsvObject) {
        $mgsvBytes = [Text.Encoding]::UTF8.GetBytes(($mgsvObject | ConvertTo-Json -Depth 6))
        $mgsvFile = [IO.File]::Open(($mgsvPath + '.tmp'), [IO.FileMode]::CreateNew, [IO.FileAccess]::Write, [IO.FileShare]::None)
        try { $mgsvFile.Write($mgsvBytes, 0, $mgsvBytes.Length); $mgsvFile.Flush() } finally { $mgsvFile.Dispose() }
        [IO.File]::Move(($mgsvPath + '.tmp'), $mgsvPath)
    }
    function Get-MgsvHash($mgsvPath, $mgsvLabel) {
        $mgsvFile = [IO.File]::Open($mgsvPath, [IO.FileMode]::Open, [IO.FileAccess]::Read, [IO.FileShare]::Read)
        $mgsvHash = [Security.Cryptography.SHA256]::Create()
        try {
            $mgsvTotal = $mgsvFile.Length
            if ($mgsvTotal -le 0) { throw 'Empty file.' }
            $mgsvBuffer = [byte[]]::new(4MB)
            $mgsvDone = [long]0
            $mgsvClock = [Diagnostics.Stopwatch]::StartNew()
            while (($mgsvRead = $mgsvFile.Read($mgsvBuffer, 0, $mgsvBuffer.Length)) -gt 0) {
                [void]$mgsvHash.TransformBlock($mgsvBuffer, 0, $mgsvRead, $mgsvBuffer, 0)
                $mgsvDone += $mgsvRead
                if ($mgsvClock.ElapsedMilliseconds -gt 750) {
                    Write-Progress -Activity $mgsvLabel -Status ('{0:N0} / {1:N0} MiB' -f ($mgsvDone / 1MB), ($mgsvTotal / 1MB)) -PercentComplete ([int][Math]::Min(100, 100 * $mgsvDone / $mgsvTotal))
                    $mgsvClock.Restart()
                }
            }
            [void]$mgsvHash.TransformFinalBlock([byte[]]@(), 0, 0)
            if ($mgsvDone -ne $mgsvTotal) { throw 'File size changed while hashing.' }
            return [BitConverter]::ToString($mgsvHash.Hash).Replace('-', '').ToLowerInvariant()
        } finally { $mgsvHash.Dispose(); $mgsvFile.Dispose(); Write-Progress -Activity $mgsvLabel -Completed }
    }
    function Wait-MgsvFile($mgsvPath, $mgsvSeconds, $mgsvMaximum) {
        $mgsvClock = [Diagnostics.Stopwatch]::StartNew()
        while (-not [IO.File]::Exists($mgsvPath)) {
            if ($mgsvClock.Elapsed.TotalSeconds -gt $mgsvSeconds) { throw 'Timed out waiting for the guest. No automatic restart will occur.' }
            if ($mgsvLaunched -and $mgsvLaunchClock.Elapsed.TotalSeconds -gt 60) {
                $mgsvClients = @(Get-Process | Where-Object { $_.ProcessName -in @('WindowsSandbox','WindowsSandboxClient','WindowsSandboxRemoteSession') })
                if ($mgsvClients.Count -eq 0) { throw 'Sandbox has closed. Existing reports and cache are preserved.' }
            }
            Start-Sleep -Seconds 1
        }
        $mgsvItem = Assert-MgsvPlain $mgsvPath
        if ($mgsvItem.PSIsContainer -or $mgsvItem.Length -le 0 -or $mgsvItem.Length -gt $mgsvMaximum) { throw 'Invalid guest report file.' }
    }
    function Assert-MgsvFiles($mgsvFiles) {
        if (@($mgsvFiles).Count -ne 10) { throw 'Wrong game file count.' }
        $mgsvSeen = @{}
        foreach ($mgsvEntry in $mgsvFiles) {
            Assert-MgsvKeys $mgsvEntry @('path','bytes','sha256')
            if ($mgsvSeen.ContainsKey($mgsvEntry.path)) { throw 'Duplicate game file.' }
            $mgsvExpected = @($mgsvExpectedFiles | Where-Object { $_.path -ceq $mgsvEntry.path })
            if ($mgsvExpected.Count -ne 1 -or -not (Test-MgsvInteger $mgsvEntry.bytes) -or $mgsvEntry.bytes -ne $mgsvExpected[0].bytes -or $mgsvEntry.sha256 -cne $mgsvExpected[0].sha256) { throw 'Unrecognized game file identity.' }
            $mgsvSeen[$mgsvEntry.path] = $true
        }
    }
    function Assert-MgsvBinding($mgsvObject, $mgsvStage, $mgsvChallenge, $mgsvTag) {

    }

        $mgsvPreloginText = [IO.File]::ReadAllText("$mgsvOutput\prelogin.gate")
        $mgsvMatch = [regex]::Match($mgsvPreloginText, ('\AWWQA1\n' + $mgsvRun + '\n([0-9a-f]{32})\nprelogin\n' + $mgsvPreloginTag + '\n1,1,1,1\n\z'))

        if ($mgsvStartupResult.result -cin @('exited_before_deadline','process_survived_deadline') -and (-not $mgsvStartupResult.process_started -or $null -eq $mgsvStartupResult.exit_code_hex -or $null -ne $mgsvStartupResult.native_error)) { throw 'Inconsistent process observation.' }
        if ($mgsvStartupResult.result -ceq 'exited_before_deadline' -and $mgsvStartupResult.close_method -cne 'not_needed') { throw 'Inconsistent early exit.' }
        if ($mgsvStartupResult.result -ceq 'process_survived_deadline' -and ($mgsvStartupResult.observed_seconds -lt 119 -or $mgsvStartupResult.close_method -cnotin @('graceful','forced'))) { throw 'Inconsistent timed observation.' }
        if ($mgsvStartupResult.result -ceq 'probe_error' -and $null -eq $mgsvStartupResult.native_error) { throw 'Missing probe error.' }
        if ($mgsvStartupResult.result -ceq 'cleanup_failed' -and (-not $mgsvStartupResult.process_started -or $mgsvStartupResult.close_method -cne 'failed')) { throw 'Inconsistent process cleanup.' }
        $mgsvSummary.result = 'startup_observation_collected'
        $mgsvSummary.startup_outcome = $mgsvStartupResult.result
        $mgsvSummary.process_started = $mgsvStartupResult.process_started
        $mgsvSummary.exit_code_hex = $mgsvStartupResult.exit_code_hex
        $mgsvSummary.native_error = $mgsvStartupResult.native_error
        Write-MgsvJson "$mgsvOutput\restart-result.json" $mgsvSummary
        Write-Output ('Saved final report: ' + "$mgsvOutput\restart-result.json")
        Write-Output 'Exit guest Steam normally. Once both helpers finish, you may close Sandbox; the host cache/reports remain.'
        Write-Output 'Upload only restart-result.json, plus any on-screen game error wording. Startup collection is not co-op proof.'
    } catch {
        if ($mgsvInput -and [IO.Directory]::Exists($mgsvInput)) {
            try { Write-MgsvJson "$mgsvInput\host-error.json" ([ordered]@{schema=1;result='host_coordinator_stopped'}) } catch { }
        }
        if ($mgsvSummary -and $mgsvOutput -and -not [IO.File]::Exists("$mgsvOutput\restart-result.json")) {
            $mgsvSummary.result = 'host_coordinator_stopped'
            try { Write-MgsvJson "$mgsvOutput\restart-result.json" $mgsvSummary } catch { }
        }
        Write-Output 'The coordinator stopped. Keep existing cache/reports; do not rerun or reconnect the guest.'
        throw
    } finally {
        if ($mgsvProfile -and [IO.File]::Exists($mgsvProfile) -and $mgsvRetired -and -not [IO.File]::Exists($mgsvRetired)) {
            try { [IO.File]::Move($mgsvProfile,$mgsvRetired) } catch { Write-Output 'Could not retire the launcher. Do not reopen it.' }
        }
    }
}
